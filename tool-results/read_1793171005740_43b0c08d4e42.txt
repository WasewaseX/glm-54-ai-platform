     1→/**
     2→ * Book Forge — the platform's long-form serialization engine.
     3→ *
     4→ * Architecture (the honest answer to "unlimited output"):
     5→ * - No single API request ever emits a whole book. Instead:
     6→ *   1. A deterministic story bible + 10-part arc (creative backbone, version-controlled)
     7→ *   2. LLM expands each part into 10 detailed chapter briefs (structured JSON)
     8→ *   3. 10 block-workers run in parallel; within a block chapters are sequential so each
     9→ *      chapter opens from the ACTUAL final words of its predecessor (rolling continuity)
    10→ *   4. Every chapter is validated for word count and shortfalls get one continuation call
    11→ *   5. Each finished chapter is checkpointed to disk + state.json; the run is resumable
    12→ *
    13→ * Output budgets per call stay inside the served model's real limits (max_tokens 4096).
    14→ */

   221→// --- Global concurrency gate: the gateway rate-limits hard, so serialize calls ---
   222→const MAX_CONCURRENT_LLM_CALLS = 1;
   223→const MIN_GAP_BETWEEN_CALLS_MS = 3_000;
   224→let activeCalls = 0;
   225→let lastCallStart = 0;
   226→const waiters: Array<() => void> = [];
   227→
   228→async function acquireSlot(): Promise<void> {
   229→  if (activeCalls < MAX_CONCURRENT_LLM_CALLS) {
   230→    activeCalls++;
   231→    return;
   232→  }
   233→  await new Promise<void>((resolve) => waiters.push(resolve));
   234→  activeCalls++;
   235→  // Global pacing: never start two calls closer than MIN_GAP_BETWEEN_CALLS_MS.
   236→  const sinceLast = Date.now() - lastCallStart;
   237→  if (sinceLast < MIN_GAP_BETWEEN_CALLS_MS) {
   238→    await sleep(MIN_GAP_BETWEEN_CALLS_MS - sinceLast);
   239→  }
   240→  lastCallStart = Date.now();
   241→}
   242→
   243→function releaseSlot(): void {
   244→  activeCalls--;
   245→  const next = waiters.shift();
   246→  if (next) next();
   247→}
   248→
   249→function isRateLimitError(err: unknown): boolean {
   250→  const msg = err instanceof Error ? err.message : String(err);
   251→  return msg.includes("429") || msg.toLowerCase().includes("too many requests");
   252→}
   253→
   254→async function callLLM(
   255→  messages: Array<{ role: "system" | "user" | "assistant"; content: string }>,
   256→  maxTokens = 4096
   257→): Promise<string> {
   258→  const client = await zai();
   259→  let lastErr: unknown = null;
   260→  await acquireSlot();
   261→  try {
   262→    let attempt = 0;
   263→    let hardFails = 0;
   264→    // Rate limits are NEVER fatal: the shared gateway can stay saturated for hours.
   265→    // Wait patiently (capped, jittered exponential backoff) and retry forever — the
   266→    // run is checkpointed, so patience costs nothing and wastes no quota.
   267→    while (true) {
   268→      attempt++;
   269→      try {
   270→        const res = (await client.chat.completions.create({
   271→          messages,
   272→          stream: false,
   273→          max_tokens: maxTokens,
   274→        })) as { choices?: Array<{ message?: { content?: string } }> };
   275→        const content = res?.choices?.[0]?.message?.content ?? "";
   276→        if (!content.trim()) throw new Error("empty completion");
   277→        return content;
   278→      } catch (err) {
   279→        lastErr = err;
   280→        const rateLimited = isRateLimitError(err);
   281→        if (rateLimited) {
   282→          const backoff = Math.min(30_000 * 2 ** (attempt - 1), 480_000); // 30s → 8 min cap
   283→          const waitMs = backoff + Math.random() * 30_000;
   284→          log(
   285→            `LLM rate-limited (retry #${attempt}): ${
   286→              err instanceof Error ? err.message : String(err)
   287→            } — waiting ${Math.round(waitMs / 1000)}s`
   288→          );
   289→          await sleep(waitMs);
   290→        } else {
   291→          hardFails++;
   292→          if (hardFails >= 5) {
   293→            throw lastErr instanceof Error ? lastErr : new Error(String(lastErr));
   294→          }
   295→          const waitMs = 5_000 * hardFails;
   296→          log(
   297→            `LLM call failed (attempt ${attempt}, non-rate-limit): ${
   298→              err instanceof Error ? err.message : String(err)
   299→            } — waiting ${Math.round(waitMs / 1000)}s`
   300→          );
   301→          await sleep(waitMs);
   302→        }
   303→      }
   304→    }
   305→  } finally {
   306→    releaseSlot();
   307→  }
   308→}
   309→
   310→function sleep(ms: number): Promise<void> {
   311→  return new Promise((r) => setTimeout(r, ms));
   312→}

   370→/** True when a runner process is alive for the book (pid file exists + pid responds). */
   371→export function isRunnerRunning(bookId: string): boolean {
   372→  const pidPath = join("/tmp", `novel-${bookId}.pid`);
   373→  if (!existsSync(pidPath)) return false;
   374→  try {
   375→    const pid = Number(readFileSync(pidPath, "utf8").trim());
   376→    process.kill(pid, 0); // throws when not alive
   377→    return true;
   378→  } catch {
   379→    return false;
   380→  }
   381→}

   383→/** Spawn the resumable runner for a book as a detached process (single-instance safe). */
   384→export function startBookRunner(bookId: string): { started: boolean; message: string } {
   385→  const def = BOOKS[bookId];
   386→  if (!def) return { started: false, message: `unknown book: ${bookId}` };
   387→  if (isRunnerRunning(bookId)) {
   388→    const pid = Number(readFileSync(join("/tmp", `novel-${bookId}.pid`), "utf8").trim());
   389→    return { started: false, message: `Runner already active (pid ${pid}).` };
   390→  }
   391→  const state = loadState(bookId);
   392→  if (state?.phase === "done") {
   393→    return { started: false, message: "Book already complete." };
   394→  }
   395→  const logPath = join("/tmp", "novel-run.log");
   396→  const out = openSync(logPath, "a");
   397→  const child = spawn("bun", ["scripts/run-novel.ts", bookId], {
   398→    cwd: ROOT,
   399→    stdio: ["ignore", out, out],
   400→    detached: true,
   401→    env: process.env,
   402→  });
   403→  child.unref();
   404→  writeFileSync(join("/tmp", `novel-${bookId}.pid`), String(child.pid));
   405→  return { started: true, message: `Book Forge started (pid ${child.pid}).` };
   406→}

   421→async function generateOutlinePart(def: BookDef, part: PartDef): Promise<void> {

   423→  const outlinePath = join(dir, "outline.json");

   425→    ? (JSON.parse(readFileSync(outlinePath, "utf8")) as Record<string, unknown>)
   426→    : {};

   428→  if (Array.isArray(outline[String(part.index)])) return; // already done
   429→
   430→  const prompt = `You are the story architect for the novel "${def.title}".
   431→
   432→STORY BIBLE (ground truth):
   433→${def.bible}
   434→
   435→PART ${part.index} OF 10 — "${part.name}" (chapters ${part.range[0]}-${part.range[1]}):
   436→${part.synopsis}
   437→The part opens from this state: ${part.openingBeat}

   439→
   440→Expand this part into EXACTLY 10 chapter briefs, one per chapter (${part.range[0]} through ${part.range[1]}).

   443→Respond with ONLY a JSON array (no markdown fences, no commentary) of 10 objects:
   444→[{"n": <chapter number>, "title": "...", "brief": "...", "hook": "..."}]`;

   446→  const raw = await callLLM(
   447→    [
   448→      { role: "system", content: "You are a rigorous story architect who outputs only valid JSON." },
   449→      { role: "user", content: prompt },
   450→    ],
   451→    4096
   452→  );

   454→  const jsonText = raw.replace(/^\s*```(?:json)?/i, "").replace(/```\s*$/i, "").trim();
   455→  const chapters = JSON.parse(jsonText) as Array<{ n: number; title: string; brief: string; hook: string }>;
   456→  if (!Array.isArray(chapters) || chapters.length !== 10) {
   457→    throw new Error(`outline part ${part.index}: expected 10 briefs, got ${chapters?.length}`);
   458→  }
   459→  outline[String(part.index)] = chapters;
   460→  writeFileSync(outlinePath, JSON.stringify(outline, null, 2));
   461→  log(`[outline] part ${part.index} (${part.name}) — 10 briefs`);
   462→}

   464→function previousChapterTail(def: BookDef, state: BookState, n: number): string {
   465→  if (n > 1) {
   466→    const prev = readChapter(def.id, n - 1);
   467→    if (prev) {
   468→      const body = prev.split("\n").slice(1).join("\n").trim(); // drop title line
   469→      return `CONTINUITY — the previous chapter (Chapter ${n - 1}) ended with these final words. Your chapter must follow them seamlessly in time and mood:\n"""\n${clipText(body, 1600)}\n"""`;
   470→    }
   471→  }
   472→  const partIdx = Math.ceil(n / 10);
   473→  const part = def.parts[partIdx - 1];
   474→  return `CONTINUITY — this chapter OPENS Part ${part.index} ("${part.name}"). The story so far stands at: ${part.openingBeat}`;
   475→}

   477→async function generateChapter(def: BookDef, state: BookState, n: number): Promise<void> {
   478→  const partIdx = Math.ceil(n / 10);
   479→  const part = def.parts[partIdx - 1];
   480→  const outline = JSON.parse(readFileSync(join(bookDir(def.id), "outline.json"), "utf8")) as Record<
   481→    string,
   482→    Array<{ n: number; title: string; brief: string; hook: string }>
   483→  >;
   484→  const brief = outline[String(partIdx)].find((c) => c.n === n);
   485→  if (!brief) throw new Error(`chapter ${n}: no brief found for part ${partIdx}`);
   486→
   487→  const cs = state.chapters[n] ?? (state.chapters[n] = { status: "pending", words: 0, attempts: 0 });
   488→  cs.attempts++;
   489→
   490→  const messages: Array<{ role: "system" | "user" | "assistant"; content: string }> = [
   491→    {
   492→      role: "system",
   493→      content: `You are the novelist writing "${def.title}".\n\n${def.bible}\n\n${STYLE_RULES}`,
   494→    },
   495→    {
   496→      role: "user",
   497→      content: `PART ${part.index} OF 10 — "${part.name}" (chapters ${part.range[0]}-${part.range[1]}):
   498→${part.synopsis}
   499→
   500→CHAPTER ${n} OF 100 — "${brief.title}"
   501→${brief.brief}
   502→The chapter must END on this hook: ${brief.hook}
   503→
   504→${previousChapterTail(def, state, n)}
   505→
   506→Write the complete chapter now, 2,100-2,300 words. First line: "Chapter ${n} — ${brief.title}" then the prose.`,
   507→    },
   508→  ];
   509→
   510→  let text = await callLLM(messages, 4096);
   511→  let words = countWords(text);
   512→
   513→  // Continuation fill if the model came back short.
   514→  if (words < 1900) {
   515→    log(`[ch${n}] short (${words} words) — requesting continuation`);
   516→    const fill = await callLLM(
   517→      [
   518→        ...messages,
   519→        { role: "assistant", content: text },
   520→        {
   521→          role: "user",
   522→          content: `The chapter above is only ${words} words and must reach 2,100-2,300. Continue it EXACTLY from its final sentence (do not repeat, do not summarize, do not restart). Write roughly ${2300 - words} more words of the same scene or its continuation, and make sure the chapter's final line lands the required hook: ${brief.hook}. Output ONLY the continuation prose.`,
   523→        },
   524→      ],
   525→      2048
   526→    );
   527→    text = `${text.trim()}\n${fill.trim()}`;
   528→    words = countWords(text);
   529→  }
   530→
   531→  if (words < 1600) throw new Error(`chapter ${n}: too short after continuation (${words} words)`);
   532→
   533→  mkdirSync(join(bookDir(def.id), "manuscript"), { recursive: true });
   534→  writeFileSync(chapterFile(def.id, n), text.trim() + "\n");
   535→  cs.status = "done";
   536→  cs.words = words;
   537→  cs.title = brief.title;
   538→  state.totalWords += words - (state.chapters[n]?.words && cs.status === "done" ? state.chapters[n].words : 0);
   539→  state.totalWords = Object.values(state.chapters)
   540→    .filter((c) => c.status === "done")
   541→    .reduce((a, c) => a + c.words, 0);
   542→  state.chaptersDone = Object.values(state.chapters).filter((c) => c.status === "done").length;
   543→  saveState(state);
   544→  log(`[ch${n}] done — ${words} words (total ${state.totalWords})`);
   545→}

   551→async function runBlock(def: BookDef, state: BookState, partIdx: number): Promise<void> {
   552→  const part = def.parts[partIdx - 1];
   553→  for (let n = part.range[0]; n <= part.range[1]; n++) {
   554→    const cs = state.chapters[n];
   555→    if (cs?.status === "done") continue; // resumable
   556→    // Rate-limit errors no longer reach this catch (callLLM waits them out forever),
   557→    // so a failure here is a genuine generation problem. Retry the SAME chapter up
   558→    // to 3 times, then defer — a later pass re-picks it (everything is checkpointed).
   559→    let strikes = 0;
   560→    while (state.chapters[n]?.status !== "done") {
   561→      try {
   562→        await generateChapter(def, state, n);
   563→        break;
   564→      } catch (err) {
   565→        strikes++;
   566→        const msg = `chapter ${n}: ${err instanceof Error ? err.message : String(err)}`;
   567→        state.errors.push(msg);
   568→        state.chapters[n] = {
   569→          status: "failed",
   570→          words: 0,
   571→          attempts: (state.chapters[n]?.attempts ?? 0) + 1,
   572→        };
   573→        saveState(state);
   574→        log(`[ch${n}] FAILED (strike ${strikes}/3) — ${msg}`);
   575→        if (strikes >= 3) {
   576→          log(`[ch${n}] deferred to a later pass`);
   577→          break;
   578→        }
   579→        await sleep(60_000);
   580→      }
   581→    }
   582→  }
   583→}

   585→/** Run the full book: outline (parallel) then 10 parallel block-workers. */
   586→export async function runBook(bookId: string, forceRestart = false): Promise<BookState> {
   587→  const def = BOOKS[bookId];
   588→  if (!def) throw new Error(`unknown book: ${bookId}`);
   589→
   590→  mkdirSync(bookDir(bookId), { recursive: true });
   591→  mkdirSync(join(bookDir(bookId), "manuscript"), { recursive: true });
   592→
   593→  let state = loadState(bookId);
   594→  if (!state || forceRestart) {
   595→    state = {
   596→      bookId,
   597→      title: def.title,
   598→      phase: "bible",
   599→      chapters: {},
   600→      totalWords: 0,
   601→      chaptersDone: 0,
   602→      errors: [],
   603→      startedAt: new Date().toISOString(),
   604→      updatedAt: new Date().toISOString(),
   605→    };
   606→    for (let n = 1; n <= 100; n++) state.chapters[n] = { status: "pending", words: 0, attempts: 0 };
   607→    writeFileSync(join(bookDir(bookId), "bible.md"), def.bible);
   608→    saveState(state);
   609→  }
   610→
   611→  if (state.phase === "done") {
   612→    log(`[run] book ${bookId} already complete`);
   613→    return state;
   614→  }
   615→
   616→  log(`[run] Book Forge starting "${def.title}" (100 chapters × ~2,200 words)`);
   617→  state.phase = "outline";
   618→  saveState(state);
   619→
   620→  // 1) Outlines — sequential and patient (quota-friendly), resumable per part.
   621→  for (const p of def.parts) {
   622→    let ok = false;
   623→    for (let attempt = 1; attempt <= 8 && !ok; attempt++) {
   624→      try {
   625→        await generateOutlinePart(def, p);
   626→        ok = true;
   627→      } catch (err) {
   628→        log(
   629→          `[outline] part ${p.index} attempt ${attempt} failed: ${
   630→            err instanceof Error ? err.message : String(err)
   631→          }`
   632→        );
   633→        await sleep(15_000 * attempt);
   634→      }
   635→    }
   636→    if (!ok) throw new Error(`outline part ${p.index} failed after 8 attempts`);
   637→  }
   638→
   639→  state.phase = "chapters";
   640→  saveState(state);
   641→
   642→  // 2) Chapters — blocks run in parallel for continuity pipelining; the global
   643→  //    call gate serializes actual LLM traffic, so quota is respected automatically.
   644→  await Promise.all(def.parts.map((p) => runBlock(def, state!, p.index)));
   645→
   646→  // 3) Final tally.
   647→  const failed = Object.entries(state.chapters)
   648→    .filter(([, c]) => c.status !== "done")
   649→    .map(([n]) => Number(n));
   650→  state.chaptersDone = Object.values(state.chapters).filter((c) => c.status === "done").length;
   651→  state.totalWords = Object.values(state.chapters)
   652→    .filter((c) => c.status === "done")
   653→    .reduce((a, c) => a + c.words, 0);
   654→  state.phase = failed.length === 0 ? "done" : "chapters"; // resumable if gaps remain
   655→  saveState(state);
   656→  log(`[run] Book Forge finished pass — ${state.chaptersDone}/100 chapters, ${state.totalWords} words, failed: [${failed.join(", ")}]`);
   657→
   658→  // Auto-deliver: the moment the book is complete, assemble .md/.txt/.pdf so the
   659→  // final output never depends on an external nudge.
   660→  if (state.phase === "done") {
   661→    try {
   662→      log("[run] assembling deliverables (.md/.txt/.pdf)…");
   663→      execFileSync("python3", ["scripts/assemble-novel.py", bookId], {
   664→        cwd: ROOT,
   665→        timeout: 300_000,
   666→      });
   667→      log("[run] deliverables assembled to public/deliverables/");
   668→    } catch (err) {
   669→      log(
   670→        `[run] auto-assembly failed (can be run manually): ${
   671→          err instanceof Error ? err.message : String(err)
   672→        }`
   673→      );
   674→    }
   675→  }
   676→
   677→  return state;
   678→}
