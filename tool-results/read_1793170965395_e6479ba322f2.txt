   108→   108→// ---------------------------------------------------------------------------
   109→   109→// Block parsing helpers
   110→   110→// ---------------------------------------------------------------------------
   111→   111→
   112→   112→type BlockKind = "plan" | "tool" | "artifact";
   113→   113→
   114→   114→export interface ParsedBlock {
   115→   115→  kind: BlockKind;
   116→   116→  attrs: Record<string, string>;
   117→   117→  content: string;
   118→   118→  complete: boolean;
   119→   119→}
   120→   120→
   121→   121→const OPENERS: Record<BlockKind, string> = {
   122→   122→  plan: "<plan",
   123→   123→  tool: "<tool",
   124→   124→  artifact: "<artifact",
   125→   125→};
   126→   126→const CLOSERS: Record<BlockKind, string> = {
   127→   127→  plan: "</plan>",
   128→   128→  tool: "</tool>",
   129→   129→  artifact: "</artifact>",
   130→   130→};
   131→   131→const BLOCK_KINDS: BlockKind[] = ["plan", "tool", "artifact"];
   132→   132→
   133→   133→function matchOpener(rest: string): BlockKind | null {
   134→   134→  for (const kind of BLOCK_KINDS) {
   135→   135→    const start = OPENERS[kind];
   136→   136→    if (!rest.startsWith(start)) continue;
   137→   137→    const next = rest[start.length];
   138→   138→    if (next === undefined) return kind; // header still streaming in
   139→   139→    if (next === ">" || /\s/.test(next)) return kind;
   140→   140→  }
   141→   141→  return null;
   142→   142→}
   143→   143→
   144→   144→/** True when the text could still become an opener once more chunks arrive. */
   145→   145→function isPartialOpener(s: string): boolean {
   146→   146→  if (s.length === 0 || s.includes(">")) return false;
   147→   147→  for (const kind of BLOCK_KINDS) {
   148→   148→    const full = OPENERS[kind];
   149→   149→    if (full.startsWith(s)) return true;
   150→   150→    if (s.startsWith(full)) {
   151→   151→      const next = s[full.length];
   152→   152→      if (next === undefined || /\s/.test(next)) return true;
   153→   153→    }
   154→   154→  }
   155→   155→  return false;
   156→   156→}
   157→   157→
   158→   158→/** Length of the longest suffix of `text` that is a proper prefix of `pattern`. */
   159→   159→function partialSuffixLength(text: string, pattern: string): number {
   160→   160→  const max = Math.min(text.length, pattern.length - 1);
   161→   161→  for (let len = max; len > 0; len--) {
   162→   162→    if (text.endsWith(pattern.slice(0, len))) return len;
   163→   163→  }
   164→   164→  return 0;
   165→   165→}
   166→   166→
   167→   167→function parseAttrs(raw: string): Record<string, string> {
   168→   168→  const attrs: Record<string, string> = {};
   169→   169→  const re = /([a-zA-Z_][\w-]*)\s*=\s*"([^"]*)"|([a-zA-Z_][\w-]*)\s*=\s*'([^']*)'/g;
   170→   170→  let m: RegExpExecArray | null;
   171→   171→  while ((m = re.exec(raw)) !== null) {
   172→   172→    attrs[m[1]] = (m[2] ?? m[4] ?? "").trim();
   173→   173→  }
   174→   174→  return attrs;
   175→   175→}
   176→   176→
   177→   177→// ---------------------------------------------------------------------------
   178→   178→// Streaming block filter
   179→   179→// ---------------------------------------------------------------------------
   180→   180→
   181→   181→/**
   182→   182→ * Streams model output to the user while suppressing plan/tool/artifact
   183→   183→ * blocks. Partial openers/closers at chunk boundaries are buffered until
   184→   184→ * resolved. Artifact bodies additionally stream out live via onArtifactDelta.
   185→   185→ */
   186→   186→export class BlockStreamFilter {
   187→   187→  private buffer = "";
   188→   188→  private block: { kind: BlockKind; attrs: Record<string, string>; content: string; sent: number } | null =
   189→   189→    null;
   190→   190→  readonly blocks: ParsedBlock[] = [];
   191→   191→  private artifactIndex = 0;
   192→   192→
   193→   193→  constructor(
   194→   194→    private readonly onVisible: (text: string) => void,
   195→   195→    private readonly onArtifactDelta: (tempId: string, text: string) => void
   196→   196→  ) {}
   197→   197→
   198→   198→  push(text: string): void {
   199→   199→    if (!text) return;
   200→   200→    this.buffer += text;
   201→   201→    this.process(false);
   202→   202→  }
   203→   203→
   204→   204→  /** Flush at end of stream; finalizes unclosed blocks. Returns all parsed blocks. */
   205→   205→  end(): ParsedBlock[] {
   206→   206→    this.process(true);
   207→   207→    if (this.block) {
   208→   208→      this.finalizeBlock(this.block.content, false);
   209→   209→      this.block = null;
   210→   210→    }
   211→   211→    if (this.buffer) {
   212→   212→      this.onVisible(this.buffer);
   213→   213→      this.buffer = "";
   214→   214→    }
   215→   215→    return this.blocks;
   216→   216→  }
   217→   217→
   218→   218→  private process(flush: boolean): void {
   219→   219→    let progress = true;
   220→   220→    while (progress) {
   221→   221→      progress = false;
   222→   222→
   223→   223→      if (this.block) {
   224→   224→        const closer = CLOSERS[this.block.kind];
   225→   225→        const closeIdx = this.buffer.indexOf(closer);
   226→   226→        if (closeIdx >= 0) {
   227→   227→          const part = this.buffer.slice(0, closeIdx);
   228→   228→          this.emitArtifactPart(part);
   229→   229→          this.buffer = this.buffer.slice(closeIdx + closer.length);
   230→   230→          this.finalizeBlock(this.block.content, true);
   231→   231→          this.block = null;
   232→   232→          progress = true;
   233→   233→        } else if (flush) {
   234→   234→          this.emitArtifactPart(this.buffer);
   235→   235→          this.buffer = "";
   236→   236→          this.finalizeBlock(this.block.content, false);
   237→   237→          this.block = null;
   238→   238→          progress = true;
   239→   239→        } else {
   240→   240→          const hold = partialSuffixLength(this.buffer, closer);
   241→   241→          const emitLen = this.buffer.length - hold;
   242→   242→          if (emitLen > 0) {
   243→   243→            const part = this.buffer.slice(0, emitLen);
   244→   244→            this.emitArtifactPart(part);
   245→   245→            this.buffer = this.buffer.slice(emitLen);
   246→   246→          }
   247→   247→        }
   248→   248→        continue;
   249→   249→      }
   250→   250→
   251→   251→      // Normal mode: find the next '<' that could start one of our openers.
   252→   252→      let lt = this.buffer.indexOf("<");
   253→   253→      if (lt === -1) {
   254→   254→        if (this.buffer) {
   255→   255→          this.onVisible(this.buffer);
   256→   256→          this.buffer = "";
   257→   257→        }
   258→   258→        continue;
   259→   259→      }
   260→   260→
   261→   261→      if (lt > 0) {
   262→   262→        this.onVisible(this.buffer.slice(0, lt));
   263→   263→        this.buffer = this.buffer.slice(lt);
   264→   264→        lt = 0;
   265→   265→      }
   266→   266→
   267→   267→      const rest = this.buffer;
   268→   268→      const opener = matchOpener(rest);
   269→   269→      if (opener) {
   270→   270→        const nameLen = OPENERS[opener].length;
   271→   271→        const gt = this.buffer.indexOf(">", nameLen);
   272→   272→        if (gt === -1) {
   273→   273→          if (flush) {
   274→   274→            // Malformed opener at EOF — treat as visible text.
   275→   275→            this.onVisible(this.buffer.slice(0, 1));
   276→   276→            this.buffer = this.buffer.slice(1);
   277→   277→            progress = true;
   278→   278→            continue;
   279→   279→          }
   280→   280→          return; // wait for the rest of the header
   281→   281→        }
   282→   282→        const attrText = this.buffer.slice(nameLen, gt);
   283→   283→        this.buffer = this.buffer.slice(gt + 1);
   284→   284→        this.block = {
   285→   285→          kind: opener,
   286→   286→          attrs: opener === "artifact" ? parseAttrs(attrText) : {},
   287→   287→          content: "",
   288→   288→          sent: 0,
   289→   289→        };
   290→   290→        if (opener === "artifact") this.artifactIndex += 1;
   291→   291→        progress = true;
   292→   292→        continue;
   293→   293→      }
   294→   294→
   295→   295→      if (!flush && isPartialOpener(rest)) return; // wait for more chunks
   296→   296→
   297→   297→      // Emit this '<' and keep scanning.
   298→   298→      this.onVisible(this.buffer.slice(0, 1));
   299→   299→      this.buffer = this.buffer.slice(1);
   300→   300→      progress = true;
   301→   301→    }
   302→   302→  }
   303→   303→
   304→   304→  private emitArtifactPart(part: string): void {
   305→   305→    if (!this.block || !part) return;
   306→   306→    this.block.content += part;
   307→   307→    if (this.block.kind === "artifact") {
   308→   308→      const unsent = this.block.content.slice(this.block.sent);
   309→   309→      if (unsent) {
   310→   310→        this.onArtifactDelta(`pending-artifact-${this.artifactIndex}`, unsent);
   311→   311→        this.block.sent = this.block.content.length;
   312→   312→      }
   313→   313→    }
   314→   314→  }
   315→   315→
   316→   316→  private finalizeBlock(content: string, complete: boolean): void {
   317→   317→    if (!this.block) return;
   318→   318→    this.blocks.push({
   319→   319→      kind: this.block.kind,
   320→   320→      attrs: this.block.attrs,
   321→   321→      content,
   322→   322→      complete,
   323→   323→    });
   324→   324→  }
   325→   325→}

   331→   331→interface ToolOutcome {

   339→   339→function clip(text: string, max: number): string {
   340→   340→  return text.length <= max ? text : text.slice(0, max) + "…";
   341→   341→}
   342→   342→
   343→   343→function formatSearchResults(results: WebSearchResult[]): string {
   344→   344→  if (results.length === 0) return "No results found.";
   345→   345→  return results
   346→   346→    .map((r, i) => {
   347→   347→      const date = r.date ? ` (${r.date})` : "";
   348→   348→      return `[${i + 1}] ${r.name}${date} — ${r.host_name}\nURL: ${r.url}\n${clip(r.snippet, 400)}`;
   349→   349→    })
   350→   350→    .join("\n\n");
   351→   351→}
   352→   352→
   353→   353→async function executeTool(
   354→   354→  userId: string,
   355→   355→  name: string,
   356→   356→  args: Record<string, unknown>
   357→   357→): Promise<ToolOutcome> {
   358→   358→  switch (name) {
   359→   359→    case "web_search": {
   360→   360→      const query = typeof args.query === "string" ? args.query.trim().slice(0, 400) : "";
   361→   361→      if (!query) return { ok: false, resultText: "web_search failed: missing query.", detail: {} };
   362→   362→      const numRaw = typeof args.num === "number" ? Math.round(args.num) : 5;
   363→   363→      const num = Math.min(10, Math.max(1, Number.isFinite(numRaw) ? numRaw : 5));
   364→   364→      const results = await webSearch(query, num);
   365→   365→      return {
   366→   366→        ok: true,
   367→   367→        resultText: `web_search("${query}") → ${results.length} results:\n\n${formatSearchResults(results)}`,
   368→   368→        detail: { query, count: results.length },
   369→   369→      };
   370→   370→    }
   371→   371→
   372→   372→    case "page_reader": {
   373→   373→      const url = typeof args.url === "string" ? args.url.trim().slice(0, 2_000) : "";
   374→   374→      if (!/^https?:\/\//i.test(url)) {
   375→   375→        return { ok: false, resultText: "page_reader failed: a valid http(s) URL is required.", detail: { url } };
   376→   376→      }
   377→   377→      const page = await pageReader(url);
   378→   378→      const text = clip(page.text || "(no extractable text)", PAGE_READER_CHAR_BUDGET);
   379→   379→      return {
   380→   380→        ok: true,
   381→   381→        resultText: `page_reader("${url}") — "${page.title}" (${text.length} chars extracted):\n\n${text}`,
   382→   382→        detail: { url, title: page.title, chars: text.length },
   383→   383→      };
   384→   384→    }
   385→   385→
   386→   386→    case "generate_image": {
   387→   387→      const prompt = typeof args.prompt === "string" ? args.prompt.trim().slice(0, 1_000) : "";
   388→   388→      if (!prompt) return { ok: false, resultText: "generate_image failed: missing prompt.", detail: {} };
   389→   389→      const size = isImageSize(args.size) ? args.size : "1024x1024";
   390→   390→      const b64 = await generateImage(prompt, size);
   391→   391→      const slug =
   392→   392→        prompt
   393→   393→          .toLowerCase()
   394→   394→          .replace(/[^a-z0-9]+/g, "-")
   395→   395→          .replace(/^-+|-+$/g, "")
   396→   396→          .slice(0, 48) || "image";
   397→   397→      const doc = await db.document.create({
   398→   398→        data: {
   399→   399→          userId,
   400→   400→          name: `${slug}.png`,
   401→   401→          mime: "image/png",
   402→   402→          size: Math.round((b64.length * 3) / 4),
   403→   403→          kind: "image",
   404→   404→          content: b64,
   405→   405→        },
   406→   406→      });
   407→   407→      return {
   408→   408→        ok: true,
   409→   409→        resultText: `generate_image ok. The illustration is saved as artifact "${slug}" and shown to the user automatically. documentId=${doc.id}`,
   410→   410→        detail: { prompt, size, documentId: doc.id },
   411→   411→        imageArtifact: { title: slug, documentId: doc.id },
   412→   412→      };
   413→   413→    }
   414→   414→
   415→   415→    case "run_code": {
   416→   416→      const code = typeof args.code === "string" ? args.code.slice(0, 100_000) : "";
   417→   417→      if (!code.trim()) return { ok: false, resultText: "run_code failed: missing code.", detail: {} };
   418→   418→      const language = args.language === "javascript" ? "javascript" : "typescript";
   419→   419→      const tests = typeof args.tests === "string" ? args.tests.slice(0, 60_000) : undefined;
   420→   420→      const stdin = typeof args.stdin === "string" ? args.stdin.slice(0, 20_000) : undefined;
   421→   421→      const timeoutMs = typeof args.timeoutMs === "number" ? Math.round(args.timeoutMs) : undefined;
   422→   422→      if (!(await ensureCodeExec())) {
   423→   423→        return { ok: false, resultText: "run_code failed: the code-exec service could not be started.", detail: {} };
   424→   424→      }
   425→   425→      try {
   426→   426→        const res = await fetch(CODE_EXEC_URL, {
   427→   427→          method: "POST",
   428→   428→          headers: { "Content-Type": "application/json" },
   429→   429→          body: JSON.stringify({ code, language, tests, stdin, timeoutMs }),
   430→   430→          signal: AbortSignal.timeout(30_000),
   431→   431→        });
   432→   432→        if (!res.ok) {
   433→   433→          const t = await res.text();
   434→   434→          return { ok: false, resultText: `run_code failed: executor HTTP ${res.status} ${clip(t, 300)}`, detail: {} };
   435→   435→        }
   436→   436→        const r = (await res.json()) as {
   437→   437→          exitCode: number;
   438→   438→          stdout: string;
   439→   439→          stderr: string;
   440→   440→          timedOut: boolean;
   441→   441→          durationMs: number;
   442→   442→          ranTests: boolean;
   443→   443→        };
   444→   444→        const verdict = r.ranTests ? "tests" : "program";
   445→   445→        const header = r.timedOut
   446→   446→          ? `run_code: ${verdict} TIMED OUT after ${r.durationMs}ms`
   447→   447→          : `run_code: ${verdict} ${r.exitCode === 0 ? "PASSED" : `FAILED (exit ${r.exitCode})`} in ${r.durationMs}ms`;
   448→   448→        const parts = [header, "", "--- stdout ---", clip(r.stdout || "(empty)", 3_500), "--- stderr ---", clip(r.stderr || "(empty)", 3_500)];
   449→   449→        return {
   450→   450→          ok: !r.timedOut && r.exitCode === 0,
   451→   451→          resultText: parts.join("\n"),
   452→   452→          detail: { exitCode: r.exitCode, timedOut: r.timedOut, durationMs: r.durationMs, ranTests: r.ranTests },
   453→   453→        };
   454→   454→      } catch (err) {
   455→   455→        return {
   456→   456→          ok: false,
   457→   457→          resultText: `run_code failed: ${err instanceof Error ? err.message : String(err)} (is the code-exec service on :3010 running?)`,
   458→   458→          detail: {},
   459→   459→        };
   460→   460→      }
   461→   461→    }
   462→   462→
   463→   463→    case "book_forge": {
   464→   464→      const action = args.action === "start" ? "start" : "status";
   465→   465→      const requested = typeof args.bookId === "string" ? args.bookId : "the-dreaming-vat";
   466→   466→      const bookId = BOOKS[requested] ? requested : "the-dreaming-vat";
   467→   467→      const def = BOOKS[bookId];
   468→   468→
   469→   469→      let startMessage = "";
   470→   470→      if (action === "start") {
   471→   471→        const r = startBookRunner(bookId);
   472→   472→        startMessage = r.message;
   473→   473→      }
   474→   474→
   475→   475→      const state = loadState(bookId);
   476→   476→      const running = isRunnerRunning(bookId);
   477→   477→      const failedChapters = state
   478→   478→        ? Object.entries(state.chapters)
   479→   479→            .filter(([, c]) => c.status === "failed")
   480→   480→            .map(([n]) => Number(n))
   481→   481→        : [];
   482→   482→
   483→   483→      const lines: string[] = [
   484→   484→        `Book Forge — "${def.title}" (bookId=${bookId})`,
   485→   485→        state
   486→   486→          ? `- phase: ${state.phase}`
   487→   487→          : "- phase: idle (no run started yet)",
   488→   488→        state
   489→   489→          ? `- progress: ${state.chaptersDone}/100 chapters, ${state.totalWords.toLocaleString("en-US")} words (target: 100 × ~2,200)`
   490→   490→          : "- progress: 0/100 chapters, 0 words",
   491→   491→        `- runner: ${running ? "RUNNING (checkpointed, continues automatically)" : "not running"}`,
   492→   492→        failedChapters.length > 0
   493→   493→          ? `- chapters deferred for retry: ${failedChapters.join(", ")}`
   494→   494→          : "- no failed chapters",
   495→   495→        `- deliverables (when phase=done): /deliverables/${bookId}.md and /deliverables/${bookId}.pdf — downloadable from the welcome screen`,
   496→   496→      ];
   497→   497→      if (startMessage) lines.push(`- start action result: ${startMessage}`);
   498→   498→      lines.push(
   499→   499→        "",
   500→   500→        "Report these REAL numbers to the user verbatim. If phase=done, congratulate and link the deliverables. If still writing, explain the run is checkpointed and finishes automatically; the welcome screen shows live progress."
   501→   501→      );
   502→   502→
   503→   503→      return {
   504→   504→        ok: true,
   505→   505→        resultText: clip(lines.join("\n"), TOOL_RESULT_CHAR_BUDGET),
   506→   506→        detail: {
   507→   507→          action,
   508→   508→          bookId,
   509→   509→          phase: state?.phase ?? "idle",
   510→   510→          chaptersDone: state?.chaptersDone ?? 0,
   511→   511→          totalWords: state?.totalWords ?? 0,
   512→   512→          running,
   513→   513→        },
   514→   514→      };
   515→   515→    }
   516→   516→
   517→   517→    default:
   518→   518→      return {
   519→   519→        ok: false,
   520→   520→        resultText: `Unknown tool "${name}". Available tools: web_search, page_reader, generate_image, run_code, book_forge.`,
   521→   521→        detail: {},
   522→   522→      };
   523→   523→  }
   524→   524→}

   530→   530→const ARTIFACT_KINDS: ArtifactKind[] = ["markdown", "code", "html", "svg", "image"];
   531→   531→
   532→   532→function normalizeArtifactInput(attrs: Record<string, string>, content: string) {
   533→   533→  const rawKind = (attrs.kind ?? "markdown").toLowerCase() as ArtifactKind;
   534→   534→  const kind: ArtifactKind = ARTIFACT_KINDS.includes(rawKind) ? rawKind : "markdown";
   535→   535→  const title = (attrs.title ?? "Untitled").replace(/\s+/g, " ").trim().slice(0, 120) || "Untitled";
   536→   536→  const language = attrs.language ? attrs.language.trim().slice(0, 40) : null;
   537→   537→  return { kind, title, language, content: content.slice(0, ARTIFACT_MAX_CHARS) };
   538→   538→}
   539→   539→
   540→   540→async function upsertArtifact(
   541→   541→  conversationId: string,
   542→   542→  messageId: string,
   543→   543→  input: { title: string; kind: ArtifactKind; language: string | null; content: string }
   544→   544→) {
   545→   545→  const existing = await db.artifact.findMany({
   546→   546→    where: { conversationId },
   547→   547→    select: { id: true, title: true },
   548→   548→  });
   549→   549→  const match = existing.find((a) => a.title.toLowerCase() === input.title.toLowerCase());
   550→   550→  if (match) {
   551→   551→    const updated = await db.artifact.update({
   552→   552→      where: { id: match.id },
   553→   553→      data: {
   554→   554→        messageId,
   555→   555→        kind: input.kind,
   556→   556→        language: input.language,
   557→   557→        content: input.content,
   558→   558→        version: { increment: 1 },
   559→   559→      },
   560→   560→    });
   561→   561→    return updated;
   562→   562→  }
   563→   563→  return db.artifact.create({
   564→   564→    data: {
   565→   565→      conversationId,
   566→   566→      messageId,
   567→   567→      title: input.title,
   568→   568→      kind: input.kind,
   569→   569→      language: input.language,
   570→   570→      content: input.content,
   571→   571→    },
   572→   572→  });
   573→   573→}

   576→   576→// The agent loop
   577→   577→// ---------------------------------------------------------------------------
   578→   578→
   579→   579→export interface AgentRunOptions {
   580→   580→  userId: string;
   581→   581→  conversationId: string;
   582→   582→  assistantMessageId: string;
   583→   583→  /** [system, ...history, user turn] — composed by the route. */
   584→   584→  baseMessages: SdkChatMessage[];
   585→   585→  thinking: boolean;
   586→   586→  signal: AbortSignal;
   587→   587→}
   588→   588→
   589→   589→export interface AgentRunStats {
   590→   590→  fullText: string;
   591→   591→  thinkingText: string;
   592→   592→  model: string | null;
   593→   593→  usage: { prompt_tokens: number; completion_tokens: number } | null;
   594→   594→  iterations: number;
   595→   595→  status: "complete" | "cancelled";
   596→   596→  toolsUsed: string[];
   597→   597→  artifactsCreated: number;
   598→   598→}
   599→   599→
   600→   600→/**
   601→   601→ * Runs the agent loop, streaming events through `send`. Persists agent steps
   602→   602→ * and artifacts as they happen. Throws on upstream failure (route handles).
   603→   603→ */
   604→   604→export async function runAgent(
   605→   605→  opts: AgentRunOptions,
   606→   606→  send: (event: AgentStreamEvent) => void
   607→   607→): Promise<AgentRunStats> {
   608→   608→  const messages: SdkChatMessage[] = [...opts.baseMessages];
   609→   609→  const startedAt = Date.now();
   610→   610→
   611→   611→  let fullText = ""; // user-visible accumulation (incl. [[artifact:id]] markers)
   612→   612→  let thinkingText = "";
   613→   613→  let model: string | null = null;
   614→   614→  let usage: { prompt_tokens: number; completion_tokens: number } | null = null;
   615→   615→  let stepIdx = 0;
   616→   616→  let iterations = 0;
   617→   617→  const toolsUsed = new Set<string>();
   618→   618→  let artifactsCreated = 0;
   619→   619→
   620→   620→  const aborted = () => opts.signal.aborted;
   621→   621→
   622→   622→  for (let iteration = 1; iteration <= AGENT_MAX_ITERATIONS; iteration++) {
   623→   623→    iterations = iteration;
   624→   624→    const isFinalIteration = iteration === AGENT_MAX_ITERATIONS;
   625→   625→    if (isFinalIteration) {
   626→   626→      messages.push({
   627→   627→        role: "user",
   628→   628→        content:
   629→   629→          "Tool budget reached. Write your final answer NOW. Do not emit any plan, tool, or artifact blocks.",
   630→   630→      });
   631→   631→    }
   632→   632→
   633→   633→    // ---- One model turn, streamed through the block filter ----
   634→   634→    const deltas: string[] = [];
   635→   635→    const artifactDeltas: Array<{ tempId: string; text: string }> = [];
   636→   636→    const filter = new BlockStreamFilter(
   637→   637→      (t) => deltas.push(t),
   638→   638→      (tempId, t) => artifactDeltas.push({ tempId, text: t })
   639→   639→    );
   640→   640→    const iterStart = fullText.length;
   641→   641→
   642→   642→    for await (const chunk of streamChat(messages, { thinking: opts.thinking, signal: opts.signal })) {
   643→   643→      if (chunk.model) model = chunk.model;
   644→   644→      if (chunk.usage) usage = chunk.usage;
   645→   645→      if (chunk.thinking) {
   646→   646→        thinkingText += chunk.thinking;
   647→   647→        send({ type: "thinking", text: chunk.thinking });
   648→   648→      }
   649→   649→      if (chunk.text) {
   650→   650→        deltas.length = 0;
   651→   651→        artifactDeltas.length = 0;
   652→   652→        filter.push(chunk.text);
   653→   653→        for (const d of deltas) {
   654→   654→          fullText += d;
   655→   655→          send({ type: "delta", text: d });
   656→   656→        }
   657→   657→        for (const a of artifactDeltas) send({ type: "artifact_delta", tempId: a.tempId, text: a.text });
   658→   658→      }
   659→   659→      if (chunk.finished || aborted()) break;
   660→   660→    }
   661→   661→
   662→   662→    deltas.length = 0;
   663→   663→    artifactDeltas.length = 0;
   664→   664→    const blocks = filter.end();
   665→   665→    for (const d of deltas) {
   666→   666→      fullText += d;
   667→   667→      send({ type: "delta", text: d });
   668→   668→    }
   669→   669→    for (const a of artifactDeltas) send({ type: "artifact_delta", tempId: a.tempId, text: a.text });
   670→   670→
   671→   671→    const visibleThisIter = fullText.slice(iterStart);
   672→   672→    let executedTools = false;
   673→   673→    const toolResults: string[] = [];
   674→   674→
   675→   675→    // ---- Execute blocks ----
   676→   676→    for (const block of blocks) {
   677→   677→      if (aborted()) break;
   678→   678→
   679→   679→      if (block.kind === "plan") {
   680→   680→        let steps: string[] = [];
   681→   681→        try {
   682→   682→          const parsed = JSON.parse(block.content.trim());
   683→   683→          if (Array.isArray(parsed?.steps)) {
   684→   684→            steps = parsed.steps
   685→   685→              .filter((s: unknown): s is string => typeof s === "string" && s.trim().length > 0)
   686→   686→              .map((s: string) => s.trim().slice(0, 120))
   687→   687→              .slice(0, 6);
   688→   688→          }
   689→   689→        } catch {
   690→   690→          /* malformed plan — ignore */
   691→   691→        }
   692→   692→        if (steps.length > 0) {
   693→   693→          const row = await db.agentStep.create({
   694→   694→            data: {
   695→   695→              messageId: opts.assistantMessageId,
   696→   696→              conversationId: opts.conversationId,
   697→   697→              idx: stepIdx++,
   698→   698→              kind: "plan",
   699→   699→              title: `Plan: ${steps.length} step${steps.length === 1 ? "" : "s"}`,
   700→   700→              detail: JSON.stringify({ steps }),
   701→   701→              status: "done",
   702→   702→            },
   703→   703→          });
   704→   704→          send({ type: "plan", steps });
   705→   705→          send({
   706→   706→            type: "step",
   707→   707→            id: row.id,
   708→   708→            idx: row.idx,
   709→   709→            kind: "plan",
   710→   710→            tool: null,
   711→   711→            title: row.title,
   712→   712→            status: "done",
   713→   713→            detail: row.detail,
   714→   714→          });
   715→   715→        }
   716→   716→        continue;
   717→   717→      }
   718→   718→
   719→   719→      if (block.kind === "tool") {
   720→   720→        executedTools = true;
   721→   721→        if (stepIdx >= AGENT_MAX_TOOL_STEPS) {
   722→   722→          toolResults.push("Tool budget exhausted — proceed without further tool calls.");
   723→   723→          continue;
   724→   724→        }
   725→   725→        let name = "unknown";
   726→   726→        let args: Record<string, unknown> = {};
   727→   727→        try {
   728→   728→          const parsed = JSON.parse(block.content.trim());
   729→   729→          if (typeof parsed?.name === "string") name = parsed.name;
   730→   730→          if (parsed?.arguments && typeof parsed.arguments === "object") args = parsed.arguments;
   731→   731→        } catch {
   732→   732→          toolResults.push(`Malformed tool call skipped: ${clip(block.content, 200)}`);
   733→   733→          continue;
   734→   734→        }
   735→   735→
   736→   736→        const titles: Record<string, string> = {
   737→   737→          web_search: "Searching the web",
   738→   738→          page_reader: "Reading the page",
   739→   739→          generate_image: "Generating an image",
   740→   740→        };
   741→   741→        const title = titles[name] ?? `Calling ${name}`;
   742→   742→
   743→   743→        const runningRow = await db.agentStep.create({
   744→   744→          data: {
   745→   745→            messageId: opts.assistantMessageId,
   746→   746→            conversationId: opts.conversationId,
   747→   747→            idx: stepIdx++,
   748→   748→            kind: "tool",
   749→   749→            tool: name,
   750→   750→            title,
   751→   751→            status: "running",
   752→   752→          },
   753→   753→        });
   754→   754→        toolsUsed.add(name);
   755→   755→        send({
   756→   756→          type: "step",
   757→   757→          id: runningRow.id,
   758→   758→          idx: runningRow.idx,
   759→   759→          kind: "tool",
   760→   760→          tool: name,
   761→   761→          title,
   762→   762→          status: "running",
   763→   763→        });
   764→   764→
   765→   765→        try {
   766→   766→          const outcome = await executeTool(opts.userId, name, args);
   767→   767→          await db.agentStep.update({
   768→   768→            where: { id: runningRow.id },
   769→   769→            data: {
   770→   770→              status: outcome.ok ? "done" : "error",
   771→   771→              detail: JSON.stringify(outcome.detail),
   772→   772→            },
   773→   773→          });
   774→   774→          send({
   775→   775→            type: "step",
   776→   776→            id: runningRow.id,
   777→   777→            idx: runningRow.idx,
   778→   778→            kind: "tool",
   779→   779→            tool: name,
   780→   780→            title,
   781→   781→            status: outcome.ok ? "done" : "error",
   782→   782→            detail: JSON.stringify(outcome.detail),
   783→   783→          });
   784→   784→          toolResults.push(clip(outcome.resultText, TOOL_RESULT_CHAR_BUDGET));
   785→   785→
   786→   786→          if (outcome.imageArtifact) {
   787→   787→            const artifact = await upsertArtifact(opts.conversationId, opts.assistantMessageId, {
   788→   788→              title: outcome.imageArtifact.title,
   789→   789→              kind: "image",
   790→   790→              language: null,
   791→   791→              content: outcome.imageArtifact.documentId,
   792→   792→            });
   793→   793→            fullText += `\n\n[[artifact:${artifact.id}]]\n\n`;
   794→   794→            artifactsCreated += 1;
   795→   795→            send({
   796→   796→              type: "artifact",
   797→   797→              id: artifact.id,
   798→   798→              title: artifact.title,
   799→   799→              kind: "image",
   800→   800→              language: null,
   801→   801→              version: artifact.version,
   802→   802→              updatedAt: artifact.updatedAt.toISOString(),
   803→   803→            });
   804→   804→            await db.agentStep.create({
   805→   805→              data: {
   806→   806→                messageId: opts.assistantMessageId,
   807→   807→                conversationId: opts.conversationId,
   808→   808→                idx: stepIdx++,
   809→   809→                kind: "artifact",
   810→   810→                tool: "image",
   811→   811→                title: `Image: ${artifact.title}`,
   812→   812→                detail: JSON.stringify({ artifactId: artifact.id }),
   813→   813→                status: "done",
   814→   814→              },
   815→   815→            });
   816→   816→          }
   817→   817→        } catch (err) {
   818→   818→          const message = err instanceof Error ? err.message : "Tool execution failed.";
   819→   819→          await db.agentStep.update({
   820→   820→            where: { id: runningRow.id },
   821→   821→            data: { status: "error", detail: JSON.stringify({ error: clip(message, 300) }) },
   822→   822→          });
   823→   823→          send({
   824→   824→            type: "step",
   825→   825→            id: runningRow.id,
   826→   826→            idx: runningRow.idx,
   827→   827→            kind: "tool",
   828→   828→            tool: name,
   829→   829→            title,
   830→   830→            status: "error",
   831→   831→            detail: JSON.stringify({ error: clip(message, 300) }),
   832→   832→          });
   833→   833→          toolResults.push(`Tool "${name}" failed: ${clip(message, 500)}. Continue without it.`);
   834→   834→        }
   835→   835→        continue;
   836→   836→      }
   837→   837→
   838→   838→      // ---- artifact ----
   839→   839→      const input = normalizeArtifactInput(block.attrs, block.content);
   840→   840→      const artifact = await upsertArtifact(opts.conversationId, opts.assistantMessageId, input);
   841→   841→      artifactsCreated += 1;
   842→   842→      fullText += `\n\n[[artifact:${artifact.id}]]\n\n`;
   843→   843→      send({
   844→   844→        type: "artifact",
   845→   845→        id: artifact.id,
   846→   846→        title: artifact.title,
   847→   847→        kind: input.kind,
   848→   848→        language: input.language,
   849→   849→        version: artifact.version,
   850→   850→        updatedAt: artifact.updatedAt.toISOString(),
   851→   851→      });
   852→   852→      await db.agentStep.create({
   853→   853→        data: {
   854→   854→          messageId: opts.assistantMessageId,
   855→   855→          conversationId: opts.conversationId,
   856→   856→          idx: stepIdx++,
   857→   857→          kind: "artifact",
   858→   858→          tool: "artifact",
   859→   859→          title: `Artifact: ${artifact.title}${artifact.version > 1 ? ` (v${artifact.version})` : ""}`,
   860→   860→          detail: JSON.stringify({ artifactId: artifact.id, kind: input.kind, version: artifact.version }),
   861→   861→          status: "done",
   862→   862→        },
   863→   863→      });
   864→   864→    }
   865→   865→
   866→   866→    if (aborted()) break;
   867→   867→
   868→   868→    // Tools were executed → feed results back and continue; otherwise done.
   869→   869→    if (!executedTools || isFinalIteration) break;
   870→   870→
   871→   871→    const trimmedVisible = visibleThisIter.trim();
   872→   872→    if (trimmedVisible) messages.push({ role: "assistant", content: clip(trimmedVisible, 8_000) });
   873→   873→    messages.push({
   874→   874→      role: "user",
   875→   875→      content:
   876→   876→        `<tool_results>\n\n${toolResults.join("\n\n---\n\n")}\n\n</tool_results>\n\n` +
   877→   877→        "Continue the task: call more tools if genuinely needed (end your turn right after the tool blocks), otherwise write the final answer now.",
   878→   878→    });
   879→   879→  }
   880→   880→
   881→   881→  return {
   882→   882→    fullText,
   883→   883→    thinkingText,
   884→   884→    model,
   885→   885→    usage,
   886→   886→    iterations,
   887→   887→    status: aborted() ? "cancelled" : "complete",
   888→   888→    toolsUsed: [...toolsUsed],
   889→   889→    artifactsCreated,
   890→   890→  };
   891→   891→}
   892→   892→
